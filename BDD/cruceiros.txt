DROP SCHEMA IF EXISTS cruceiros CASCADE;
CREATE SCHEMA IF NOT EXISTS cruceiros;

CREATE TABLE IF NOT EXISTS cruceiros.barcos (
    id_barco INT PRIMARY KEY,
    nome VARCHAR(100) NOT NULL,
    modelo VARCHAR(50),
    ano_fabricacion INT,
    capacidade_pasaxeros INT
);

-- Distribuir por 'id_barco'
SELECT create_distributed_table('cruceiros.barcos', 'id_barco');

CREATE TABLE IF NOT EXISTS cruceiros.portos (
    id_porto INT PRIMARY KEY,
    sku_code VARCHAR(50) NOT NULL UNIQUE,
    nome VARCHAR(100) NOT NULL,
    pais VARCHAR(50) NOT NULL,
    tasa_portuaria NUMERIC(5,2)
);

-- Convertir a Tabla de Referencia
SELECT create_reference_table('cruceiros.portos');

CREATE TABLE IF NOT EXISTS cruceiros.viaxes (
    id_viaxe BIGSERIAL NOT NULL,
    id_barco INT NOT NULL,
    id_porto INT NOT NULL,
    fecha_saida DATE,
    fecha_chegada DATE,
    
    CONSTRAINT viaxes_pkey PRIMARY KEY (id_barco, id_viaxe),
    CONSTRAINT viaxes_barco_fk FOREIGN KEY (id_barco) REFERENCES cruceiros.barcos(id_barco),
    CONSTRAINT viaxes_porto_fk FOREIGN KEY (id_porto) REFERENCES cruceiros.portos(id_porto)
);

-- Distribuir por 'id_barco'
SELECT create_distributed_table (
	'cruceiros.viaxes',
	'id_barco',
	colocate_with => 'cruceiros.barcos'
);

CREATE TABLE IF NOT EXISTS cruceiros.reservas (
    id_reserva BIGSERIAL NOT NULL,
    id_viaxe BIGINT NOT NULL,
    id_barco INT NOT NULL,
    fecha_reserva DATE DEFAULT CURRENT_DATE,
    nome_cliente VARCHAR(100),
    total_pagado NUMERIC(10, 2),
    CONSTRAINT reservas_pkey PRIMARY KEY (id_reserva, id_barco),
    CONSTRAINT reservas_viaxe_fk FOREIGN KEY (id_barco, id_viaxe)
        REFERENCES cruceiros.viaxes(id_barco, id_viaxe)
);

-- Distribuir por 'id_barco'
SELECT create_distributed_table(
    'cruceiros.reservas',
    'id_barco',
    colocate_with => 'cruceiros.viaxes'
);

SELECT
  b.id_barco,
  b.nome AS barco,
  COUNT(DISTINCT v.id_viaxe) AS total_viaxes,
  COUNT(r.id_reserva) AS total_reservas,
  COALESCE(SUM(r.total_pagado), 0) AS ingresos_totais,
  AVG(v.fecha_chegada - v.fecha_saida) AS duracion_media_dias
FROM cruceiros.barcos b
JOIN cruceiros.viaxes v
  ON v.id_barco = b.id_barco
LEFT JOIN cruceiros.reservas r
  ON r.id_barco = v.id_barco
 AND r.id_viaxe = v.id_viaxe
GROUP BY b.id_barco, b.nome
ORDER BY ingresos_totais DESC;
